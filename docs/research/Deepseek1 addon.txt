3. resources/lib/dashboard.py (GUI Dashboard)

```python
import xbmc
import xbmcgui
import xbmcaddon
import os
import subprocess
import threading
from datetime import datetime

class FluidDevDashboard:
    def __init__(self):
        self.addon = xbmcaddon.Addon()
        self.addon_path = self.addon.getAddonInfo('path')
        self.tools_path = os.path.join(self.addon_path, 'tools')
        self.reports_path = "/sdcard/Documents/EricTiemes/Tha-G-Thang/share"
        
        # Ensure tools directory exists
        os.makedirs(self.tools_path, exist_ok=True)
        os.makedirs(self.reports_path, exist_ok=True)
    
    def run(self):
        """Main dashboard loop"""
        while True:
            choice = self.show_main_menu()
            
            if choice == -1:
                break
            elif choice == 0:
                self.run_lint_scanner()
            elif choice == 1:
                self.run_ruff_analyzer()
            elif choice == 2:
                self.run_tree_generator()
            elif choice == 3:
                self.run_all_tools()
            elif choice == 4:
                self.view_reports()
            elif choice == 5:
                self.settings()
    
    def show_main_menu(self):
        """Display main menu"""
        menu_items = [
            "🔍 Quick Lint Scan",
            "🚀 Ruff Analysis", 
            "📁 Project Tree",
            "⚡ Run All Tools",
            "📄 View Reports",
            "⚙️ Settings",
            "❌ Exit"
        ]
        
        dialog = xbmcgui.Dialog()
        return dialog.select("FluidDev Tools Dashboard", menu_items)
    
    def run_lint_scanner(self):
        """Run lint scanner with progress dialog"""
        progress = xbmcgui.DialogProgress()
        progress.create("FluidDev Tools", "Running Lint Scanner...")
        
        target = self.get_target_path()
        if not target:
            return
        
        report_file = os.path.join(self.reports_path, f"lint_scan_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt")
        
        # Run in background thread
        def run_scan():
            script_path = os.path.join(self.tools_path, 'lint_scanner.py')
            cmd = ['python', script_path, target, report_file]
            
            try:
                result = subprocess.run(cmd, capture_output=True, text=True)
                self.show_result("Lint Scan Complete", report_file, result.stdout)
            except Exception as e:
                self.show_error(f"Lint scan failed: {str(e)}")
        
        thread = threading.Thread(target=run_scan)
        thread.start()
        
        # Show progress while running
        progress.update(50, "Scanning Python files...")
        thread.join()
        progress.close()
    
    def run_ruff_analyzer(self):
        """Run Ruff analysis"""
        target = self.get_target_path()
        if not target:
            return
        
        # Check if Ruff is installed
        try:
            subprocess.run(['ruff', '--version'], capture_output=True, check=True)
        except:
            if xbmcgui.Dialog().yesno("Ruff Not Found", "Ruff is not installed. Install it?"):
                self.install_ruff()
            return
        
        report_file = os.path.join(self.reports_path, f"ruff_analysis_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt")
        
        # Run Ruff
        cmd = ['ruff', 'check', target, '--output-format', 'text']
        
        progress = xbmcgui.DialogProgress()
        progress.create("FluidDev Tools", "Running Ruff Analysis...")
        
        try:
            result = subprocess.run(cmd, capture_output=True, text=True)
            
            # Save report
            with open(report_file, 'w') as f:
                f.write(f"Ruff Analysis Report\n")
                f.write(f"Generated: {datetime.now()}\n")
                f.write(f"Target: {target}\n")
                f.write("="*60 + "\n\n")
                f.write(result.stdout if result.stdout else "No issues found!")
            
            self.show_result("Ruff Analysis Complete", report_file, result.stdout[:500])
        except Exception as e:
            self.show_error(f"Ruff analysis failed: {str(e)}")
        finally:
            progress.close()
    
    def run_tree_generator(self):
        """Generate project tree"""
        target = self.get_target_path()
        if not target:
            return
        
        report_file = os.path.join(self.reports_path, f"project_tree_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt")
        
        # Generate tree
        tree_output = self.generate_tree(target)
        
        with open(report_file, 'w') as f:
            f.write(f"Project Tree\n")
            f.write(f"Generated: {datetime.now()}\n")
            f.write(f"Target: {target}\n")
            f.write("="*60 + "\n\n")
            f.write(tree_output)
        
        # Show in dialog (truncated if too long)
        if len(tree_output) > 1000:
            tree_output = tree_output[:1000] + "\n... (truncated - see full report)"
        
        self.show_result("Project Tree Generated", report_file, tree_output)
    
    def run_all_tools(self):
        """Run all tools sequentially"""
        if not xbmcgui.Dialog().yesno("Run All Tools", "This will run all analysis tools. Continue?"):
            return
        
        target = self.get_target_path()
        if not target:
            return
        
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        reports = []
        
        # 1. Lint Scanner
        progress = xbmcgui.DialogProgress()
        progress.create("FluidDev Tools", "Running All Tools...", "Step 1/3: Lint Scanner")
        
        lint_report = os.path.join(self.reports_path, f"full_scan_lint_{timestamp}.txt")
        self.run_tool("lint_scanner.py", target, lint_report)
        reports.append(f"Lint Scan: {lint_report}")
        progress.update(33, "Step 2/3: Ruff Analysis")
        
        # 2. Ruff Analysis
        ruff_report = os.path.join(self.reports_path, f"full_scan_ruff_{timestamp}.txt")
        self.run_tool_ruff(target, ruff_report)
        reports.append(f"Ruff Analysis: {ruff_report}")
        progress.update(66, "Step 3/3: Project Tree")
        
        # 3. Tree Generator
        tree_report = os.path.join(self.reports_path, f"full_scan_tree_{timestamp}.txt")
        self.run_tool_tree(target, tree_report)
        reports.append(f"Project Tree: {tree_report}")
        
        progress.update(100, "Complete!")
        progress.close()
        
        # Show summary
        summary = "All tools completed!\n\nReports saved:\n" + "\n".join(reports)
        xbmcgui.Dialog().ok("Complete!", summary)
    
    def view_reports(self):
        """Browse and view saved reports"""
        if not os.path.exists(self.reports_path):
            xbmcgui.Dialog().ok("No Reports", "No reports found yet.")
            return
        
        # List report files
        reports = []
        for f in os.listdir(self.reports_path):
            if f.endswith('.txt'):
                reports.append(f)
        
        if not reports:
            xbmcgui.Dialog().ok("No Reports", "No reports found yet.")
            return
        
        reports.sort(reverse=True)  # Newest first
        choice = xbmcgui.Dialog().select("Select Report", reports)
        
        if choice >= 0:
            report_path = os.path.join(self.reports_path, reports[choice])
            self.show_report(report_path)
    
    def show_report(self, report_path):
        """Display report in dialog"""
        try:
            with open(report_path, 'r') as f:
                content = f.read()
            
            # Truncate if too long
            if len(content) > 10000:
                content = content[:10000] + "\n... (truncated)"
            
            # Show in text viewer
            dialog = xbmcgui.Dialog()
            dialog.textviewer(f"Report: {os.path.basename(report_path)}", content)
        except Exception as e:
            self.show_error(f"Cannot read report: {str(e)}")
    
    def settings(self):
        """Addon settings"""
        options = [
            "Set Default Target Path",
            "Install/Update Ruff",
            "Clear Reports",
            "About"
        ]
        
        choice = xbmcgui.Dialog().select("Settings", options)
        
        if choice == 0:
            self.set_target_path()
        elif choice == 1:
            self.install_ruff()
        elif choice == 2:
            self.clear_reports()
        elif choice == 3:
            self.show_about()
    
    def get_target_path(self):
        """Get target path from settings or prompt"""
        target = self.addon.getSetting('default_target')
        
        if not target or not os.path.exists(target):
            # Prompt for path
            keyboard = xbmc.Keyboard("/sdcard/Documents/EricTiemes/Tha-G-Thang/public/FluidDev", "Enter Target Path")
            keyboard.doModal()
            
            if keyboard.isConfirmed():
                target = keyboard.getText()
                if os.path.exists(target):
                    self.addon.setSetting('default_target', target)
                else:
                    xbmcgui.Dialog().ok("Invalid Path", f"Path does not exist:\n{target}")
                    return None
        
        return target
    
    def set_target_path(self):
        """Set default target path"""
        keyboard = xbmc.Keyboard(self.addon.getSetting('default_target'), "Enter Default Target Path")
        keyboard.doModal()
        
        if keyboard.isConfirmed():
            new_path = keyboard.getText()
            if os.path.exists(new_path):
                self.addon.setSetting('default_target', new_path)
                xbmcgui.Dialog().ok("Success", f"Default path set to:\n{new_path}")
            else:
                xbmcgui.Dialog().ok("Error", f"Path does not exist:\n{new_path}")
    
    def install_ruff(self):
        """Install Ruff via pip"""
        if not xbmcgui.Dialog().yesno("Install Ruff", "Install Ruff linter via pip?"):
            return
        
        progress = xbmcgui.DialogProgress()
        progress.create("FluidDev Tools", "Installing Ruff...")
        
        try:
            # Try to install
            result = subprocess.run(
                ['pip', 'install', 'ruff', '--user'],
                capture_output=True,
                text=True
            )
            
            progress.close()
            
            if result.returncode == 0:
                xbmcgui.Dialog().ok("Success", "Ruff installed successfully!")
            else:
                xbmcgui.Dialog().ok("Error", f"Installation failed:\n{result.stderr}")
        except Exception as e:
            progress.close()
            self.show_error(f"Installation error: {str(e)}")
    
    def clear_reports(self):
        """Clear all saved reports"""
        if not xbmcgui.Dialog().yesno("Clear Reports", "Delete all saved reports?"):
            return
        
        try:
            count = 0
            for f in os.listdir(self.reports_path):
                if f.endswith('.txt'):
                    os.remove(os.path.join(self.reports_path, f))
                    count += 1
            
            xbmcgui.Dialog().ok("Cleared", f"Deleted {count} report(s)")
        except Exception as e:
            self.show_error(f"Failed to clear reports: {str(e)}")
    
    def show_about(self):
        """Show about dialog"""
        about_text = f"""
FluidDev Tools Dashboard v1.0

Python Development Tools for Kodi

Features:
• Lint Scanner - Find common Python issues
• Ruff Analysis - Professional Python linting
• Project Tree - Visual directory structure
• Report Viewer - Browse saved reports

Target Path: {self.addon.getSetting('default_target')}

Created by EricTiemes
        """
        
        xbmcgui.Dialog().textviewer("About FluidDev Tools", about_text)
    
    def generate_tree(self, path):
        """Generate directory tree"""
        import os
        
        def _tree(dir_path, prefix=""):
            result = []
            try:
                entries = sorted(os.listdir(dir_path))
            except:
                return ["[Permission Denied]"]
            
            # Filter out hidden/system files
            entries = [e for e in entries if not e.startswith('.') and e not in ['__pycache__']]
            
            for i, entry in enumerate(entries):
                is_last = i == len(entries) - 1
                connector = "└── " if is_last else "├── "
                full_path = os.path.join(dir_path, entry)
                
                if os.path.isdir(full_path):
                    result.append(f"{prefix}{connector}📁 {entry}/")
                    new_prefix = prefix + ("    " if is_last else "│   ")
                    result.extend(_tree(full_path, new_prefix))
                else:
                    size = os.path.getsize(full_path)
                    size_str = f" ({size:,} bytes)" if size > 0 else ""
                    result.append(f"{prefix}{connector}📄 {entry}{size_str}")
            
            return result
        
        tree_lines = [f"📁 {os.path.basename(path)}/"] + _tree(path)
        return "\n".join(tree_lines)
    
    def run_tool(self, tool_name, target, output):
        """Run a Python tool"""
        script_path = os.path.join(self.tools_path, tool_name)
        
        if not os.path.exists(script_path):
            # Create minimal script if missing
            with open(script_path, 'w') as f:
                f.write(f'print("Tool {tool_name} would run on {target}")')
        
        cmd = ['python', script_path, target, output]
        
        try:
            subprocess.run(cmd, capture_output=True, text=True)
            return True
        except:
            return False
    
    def run_tool_ruff(self, target, output):
        """Run Ruff tool"""
        cmd = ['ruff', 'check', target, '--output-format', 'text']
        
        try:
            result = subprocess.run(cmd, capture_output=True, text=True)
            with open(output, 'w') as f:
                f.write(result.stdout if result.stdout else "No issues found")
            return True
        except:
            return False
    
    def run_tool_tree(self, target, output):
        """Generate tree"""
        tree_output = self.generate_tree(target)
        with open(output, 'w') as f:
            f.write(tree_output)
        return True
    
    def show_result(self, title, report_path, content):
        """Show result with options"""
        dialog = xbmcgui.Dialog()
        
        # Truncate for dialog
        display_content = content
        if len(content) > 1000:
            display_content = content[:1000] + "\n... (truncated)"
        
        # Show result
        dialog.textviewer(title, f"{display_content}\n\nReport saved to:\n{report_path}")
    
    def show_error(self, message):
        """Show error dialog"""
        xbmcgui.Dialog().ok("Error", message)
```
