```python
import xbmc
import xbmcgui
import xbmcaddon
import sys
import os

# Add lib folder to path
ADDON = xbmcaddon.Addon()
ADDON_PATH = ADDON.getAddonInfo('path')
LIB_PATH = os.path.join(ADDON_PATH, 'resources', 'lib')
sys.path.insert(0, LIB_PATH)

# Import dashboard
try:
    from dashboard import FluidDevDashboard
    dashboard = FluidDevDashboard()
    dashboard.run()
except Exception as e:
    xbmcgui.Dialog().ok("FluidDev Error", f"Failed to start dashboard: {str(e)}")
    xbmc.log(f"FluidDev Error: {str(e)}", xbmc.LOGERROR)
```

6. Alternative: Create installer script

Create install_fluiddashboard.sh:

```bash
#!/bin/bash
# FluidDev Dashboard Installer

echo "🚀 FluidDev Tools Dashboard Installer"
echo "======================================"

# Create addon folder in Kodi directory
KODI_ADDONS="/storage/emulated/0/Android/data/org.xbmc.kodi/files/.kodi/addons"

if [ ! -d "$KODI_ADDONS" ]; then
    echo "Kodi addons folder not found!"
    echo "Please ensure Kodi is installed."
    exit 1
fi

ADDON_DIR="$KODI_ADDONS/plugin.program.fluiddashboard"

echo "📦 Installing to: $ADDON_DIR"

# Remove old if exists
rm -rf "$ADDON_DIR"

# Copy from current location
cp -r plugin.program.fluiddashboard "$KODI_ADDONS/"

echo "✅ Installation complete!"
echo ""
echo "To use:"
echo "1. Restart Kodi"
echo "2. Go to: Programs → FluidDev Tools"
echo "3. Select a tool and run!"
```

7. Quick One-Click Creation Script

Create create_fluiddashboard.sh in your scripts folder:

```bash
#!/bin/bash
# Create FluidDev Dashboard addon

BASE_DIR="/sdcard/Documents/EricTiemes/Tha-G-Thang"
ADDON_DIR="$BASE_DIR/plugin.program.fluiddashboard"
TOOLS_SRC="$BASE_DIR/share/scripts"

echo "🔨 Building FluidDev Dashboard..."

# Clean and create structure
rm -rf "$ADDON_DIR"
mkdir -p "$ADDON_DIR/resources/lib"
mkdir -p "$ADDON_DIR/tools"

# Create addon.xml
cat > "$ADDON_DIR/addon.xml" << 'EOF'
<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
<addon id="plugin.program.fluiddashboard" name="FluidDev Tools" version="1.0.0" provider-name="EricTiemes">
    <requires>
        <import addon="xbmc.python" version="3.0.0"/>
    </requires>
    <extension point="xbmc.python.pluginsource" library="default.py">
        <provides>executable</provides>
    </extension>
    <extension point="xbmc.addon.metadata">
        <summary>Python Development Tools Dashboard</summary>
        <description>Dashboard for running Python linting, analysis, and development tools directly in Kodi</description>
        <platform>all</platform>
        <license>MIT</license>
        <website>https://github.com/EricTiemes</website>
        <assets>
            <icon>icon.png</icon>
            <fanart>fanart.jpg</fanart>
        </assets>
    </extension>
</addon>
EOF

# Create default.py
cat > "$ADDON_DIR/default.py" << 'EOF'
import xbmc
import xbmcgui
import xbmcaddon
import sys
import os

ADDON = xbmcaddon.Addon()
ADDON_PATH = ADDON.getAddonInfo('path')
LIB_PATH = os.path.join(ADDON_PATH, 'resources', 'lib')
sys.path.insert(0, LIB_PATH)

try:
    from dashboard import FluidDevDashboard
    dashboard = FluidDevDashboard()
    dashboard.run()
except Exception as e:
    xbmcgui.Dialog().ok("FluidDev Error", f"Failed: {str(e)}")
EOF

# Copy dashboard.py (simplified version)
cat > "$ADDON_DIR/resources/lib/dashboard.py" << 'EOF'
import xbmc
import xbmcgui
import xbmcaddon
import os

class FluidDevDashboard:
    def __init__(self):
        self.addon = xbmcaddon.Addon()
    
    def run(self):
        menu = [
            "🔍 Quick Lint Scan",
            "🚀 Ruff Analysis", 
            "📁 Project Tree",
            "⚡ Run All Tools",
            "📄 View Reports",
            "⚙️ Settings"
        ]
        
        while True:
            choice = xbmcgui.Dialog().select("FluidDev Tools", menu)
            
            if choice == -1:
                break
            elif choice == 0:
                xbmcgui.Dialog().ok("Info", "Lint Scanner would run here")
            elif choice == 1:
                xbmcgui.Dialog().ok("Info", "Ruff Analysis would run here")
            elif choice == 2:
                xbmcgui.Dialog().ok("Info", "Tree Generator would run here")
            elif choice == 3:
                xbmcgui.Dialog().ok("Info", "All tools would run")
            elif choice == 4:
                xbmcgui.Dialog().ok("Info", "Report viewer")
            elif choice == 5:
                xbmcgui.Dialog().ok("Settings", "Settings would open here")
EOF

# Create empty init
touch "$ADDON_DIR/resources/lib/__init__.py"

# Copy your tools
cp "$TOOLS_SRC/lint_scanner.py" "$ADDON_DIR/tools/"
cp "$TOOLS_SRC/ruff_runner.py" "$ADDON_DIR/tools/"
cp "$TOOLS_SRC/tree_generator.py" "$ADDON_DIR/tools/"
cp "$TOOLS_SRC/lint_all.py" "$ADDON_DIR/tools/"

# Create placeholder images
echo "iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAYAAAAfFcSJAAAADUlEQVR42mNk+M9QDwADhgGAWjR9awAAAABJRU5ErkJggg==" | base64 -d > "$ADDON_DIR/icon.png"
echo "iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAYAAAAfFcSJAAAADUlEQVR42mNk+M9QDwADhgGAWjR9awAAAABJRU5ErkJggg==" | base64 -d > "$ADDON_DIR/fanart.jpg"

# Create ZIP
cd "$BASE_DIR"
zip -r "fluiddashboard_$(date +%Y%m%d_%H%M%S).zip" "plugin.program.fluiddashboard/"

echo "✅ Addon created!"
echo "📦 ZIP: $BASE_DIR/fluiddashboard_*.zip"
echo ""
echo "To install in Kodi:"
echo "1. Copy ZIP to device"
echo "2. Kodi → Add-ons → Install from zip"
echo "3. Browse to ZIP file"
echo "4. Install and enjoy!"
```
