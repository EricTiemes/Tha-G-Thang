## **Module: `ui_interceptor.py`**

**Captures all Kodi UI calls:**

```python
import xbmcgui

class UIInterceptor:
    def __init__(self, config):
        self.config = config
        self.original_dialog = xbmcgui.Dialog
        
    def start_monitoring(self):
        # Wrap xbmcgui.Dialog
        xbmcgui.Dialog = self._wrap_dialog(self.original_dialog)
        
    def _wrap_dialog(self, original_cls):
        interceptor = self
        
        class WrappedDialog(original_cls):
            def ok(self, heading, message):
                interceptor._log_ui_call('ok', {
                    'heading': heading,
                    'message': message
                })
                return super().ok(heading, message)
                
            def notification(self, heading, message, icon=None, time=5000):
                interceptor._log_ui_call('notification', {
                    'heading': heading,
                    'message': message,
                    'icon': icon,
                    'time': time
                })
                return super().notification(heading, message, icon, time)
                
            def progress(self):
                instance = super().progress()
                return interceptor._wrap_progress(instance)
                
        return WrappedDialog
        
    def _wrap_progress(self, progress_instance):
        """Wrap DialogProgress to log update() calls"""
        # Similar wrapping for update(), close(), etc.
        pass
        
    def _log_ui_call(self, method, params):
        self.config.log({
            'type': 'ui_call',
            'method': method,
            'params': params,
            'addon': self._get_calling_addon()
        })
```
