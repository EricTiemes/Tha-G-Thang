### 1.2 Reddit Extractor (PRAW + Fallback)
```python
# extractors/reddit.py
import praw
import requests
from bs4 import BeautifulSoup

def search_reddit(query, limit=20, sort='relevance'):
    try:
        # Primary: PRAW (optional user auth for higher limits)
        reddit = praw.Reddit(client_id='YOUR_ID', 
                             client_secret='SECRET',
                             user_agent='FluidSearch/1.0')
        subreddit = reddit.subreddit('all')
        results = subreddit.search(query, limit=limit, sort=sort)
    except:
        # Fallback: Public JSON API (no auth)
        url = f"https://www.reddit.com/search.json?q={query}&limit={limit}"
        results = requests.get(url, headers={'User-Agent': 'FluidSearch'}).json()
        results = results['data']['children']
    
    return normalize_reddit(results)
```

**No login option:** Reddit's JSON API works without auth (rate-limited but usable)

### 1.3 Deduplication (FuzzyWuzzy)
```python
# processors/deduper.py
from fuzzywuzzy import fuzz

def deduplicate(results, threshold=85):
    seen = []
    unique = []
    
    for item in results:
        is_dupe = any(
            fuzz.token_sort_ratio(item['title'], s['title']) > threshold
            for s in seen
        )
        if not is_dupe:
            unique.append(item)
            seen.append(item)
    
    return unique
```

## PHASE 2: INTELLIGENT PROCESSING

### 2.1 Scoring Algorithm
```python
# processors/scoring.py
def score_result(item, query_terms, user_prefs=None):
    score = 0
    title_lower = item['title'].lower()
    
    # Exact query match
    if all(term in title_lower for term in query_terms):
        score += 50
    
    # Partial match
    score += sum(15 for term in query_terms if term in title_lower)
    
    # Recency (max +30)
    if item['age_days'] < 7: score += 30
    elif item['age_days'] < 30: score += 20
    elif item['age_days'] < 90: score += 10
    
    # Popularity (normalized)
    if item['source'] == 'YouTube':
        score += min(item['views'] / 10000, 30)
    elif item['source'] == 'Reddit':
        score += min(item['score'] / 100, 30)
    
    # Source quality
    verified = ['official', 'verified', 'docs']
    if any(v in title_lower for v in verified):
        score += 15
    
    # Duration preference (if user set)
    if user_prefs and 'max_duration' in user_prefs:
        if item.get('duration', 0) > user_prefs['max_duration']:
            score -= 20
    
    return round(score, 2)
```

### 2.2 Auto-Summarization
```python
# processors/summarizer.py
from collections import Counter
import re

def extract_summary(text, max_sentences=3):
    sentences = re.split(r'[.!?]+', text)
    return '. '.join(sentences[:max_sentences]).strip() + '.'

def extract_keywords(results, top_n=10):
    all_text = ' '.join(r['title'] + ' ' + r.get('description', '') 
                        for r in results)
    words = re.findall(r'\b[a-z]{4,}\b', all_text.lower())
    
    stopwords = {'with', 'that', 'this', 'from', 'have', 'your'}
    filtered = [w for w in words if w not in stopwords]
    
    return [word for word, _ in Counter(filtered).most_common(top_n)]

def generate_digest(results, query):
    keywords = extract_keywords(results)
    
    digest = f"=== SEARCH DIGEST: {query} ===\n\n"
    digest += f"Found {len(results)} results\n"
    digest += f"Common themes: {', '.join(keywords[:5])}\n\n"
    
    # Top 3 by score
    top_3 = sorted(results, key=lambda x: x['score'], reverse=True)[:3]
    for i, item in enumerate(top_3, 1):
        digest += f"{i}. [{item['source']}] {item['title']}\n"
        digest += f"   {item['url']}\n"
        digest += f"   {extract_summary(item.get('description', ''))}\n\n"
    
    return digest
```

```python
# exporters/bundle.py
def export_ai_bundle(results, query, format='markdown'):
    if format == 'markdown':
        output = f"# Search Results: {query}\n\n"
        output += f"**Query:** {query}\n"
        output += f"**Sources:** {', '.join(set(r['source'] for r in results))}\n"
        output += f"**Total Results:** {len(results)}\n\n"
        
        output += "## Top Results\n\n"
        for r in results[:20]:
            output += f"### {r['title']}\n"
            output += f"- **Source:** {r['source']}\n"
            output += f"- **URL:** {r['url']}\n"
            output += f"- **Score:** {r['score']}\n"
            output += f"- **Summary:** {extract_summary(r.get('description', ''))}\n\n"
        
        output += "## Keywords\n"
        output += ', '.join(extract_keywords(results))
        
    elif format == 'json':
        output = json.dumps({
            'query': query,
            'results': results,
            'keywords': extract_keywords(results),
            'timestamp': datetime.now().isoformat()
        }, indent=2)
    
    save_path = f"{SHARE}/search_{sanitize(query)}_{timestamp()}.{format}"
    write_file(save_path, output)
    return save_path
```

```python
# exporters/playlist.py
def export_playlist(results, name):
    m3u = "#EXTM3U\n"
    
    video_results = [r for r in results if r['source'] == 'YouTube']
    
    for item in video_results:
        m3u += f"#EXTINF:{item['duration']},{item['title']}\n"
        m3u += f"{item['url']}\n"
    
    path = f"{SHARE}/playlist_{sanitize(name)}.m3u"
    write_file(path, m3u)
    return path
```

## PHASE 5: ADVANCED FEATURES

### 5.1 Smart Search (Recurring)
```python
# service.py integration
class SmartSearch:
    def __init__(self, query, filters, schedule='weekly'):
        self.query = query
        self.filters = filters
        self.schedule = schedule
        self.last_run = None
        self.seen_urls = set()
    
    def run(self):
        results = execute_search(self.query, self.filters)
        new = [r for r in results if r['url'] not in self.seen_urls]
        
        if new:
            notify(f"Fluid Search: {len(new)} new results for '{self.query}'")
            export_ai_bundle(new, f"{self.query}_new")
            self.seen_urls.update(r['url'] for r in new)
        
        self.last_run = datetime.now()
```

### 5.2 Cross-Reference Detection
```python
def find_cross_references(results):
    youtube = [r for r in results if r['source'] == 'YouTube']
    reddit = [r for r in results if r['source'] == 'Reddit']
    
    matches = []
    for yt in youtube:
        for rd in reddit:
            # Check if Reddit discusses this video
            if yt['url'] in rd.get('description', ''):
                matches.append({
                    'video': yt,
                    'discussion': rd,
                    'type': 'direct_link'
                })
            # Or similar title
            elif fuzz.ratio(yt['title'], rd['title']) > 70:
                matches.append({
                    'video': yt,
                    'discussion': rd,
                    'type': 'similar_topic'
                })
    
    return matches
```

```python
# cache.py using sqlite-utils
from sqlite_utils import Database

db = Database("search_cache.db")
db["results"].enable_fts(["title", "description"])

def cached_search(query, max_age_hours=24):
    cached = db.execute("""
        SELECT * FROM results 
        WHERE query = ? AND timestamp > datetime('now', '-{} hours')
    """, [query, max_age_hours]).fetchall()
    
    if cached: return cached
    # Else: fresh search + cache
```
