```yaml
# Code Lens: Technical Implementation

## Kodi Addon Patterns

### Structure
```

addon.py / default.py → router → resources/lib/
addon.xml → metadata & dependencies
resources/settings.xml → user config

```

### Common APIs
- Settings: `xbmcaddon.Addon().getSetting(id)`
- Logging: `xbmc.log(msg, xbmc.LOGDEBUG)`
- Dialogs: `xbmcgui.Dialog().ok(heading, message)`
- Files: `xbmcvfs.exists(path)` (not os.path)

### Termux/Python Specifics
- Unicode: always encode/decode explicit (UTF-8)
- Paths: use `os.path.join()`, never hardcode `/` or `\\`
- Permissions: check `termux-setup-storage` for external
- Modules: keep `lib/` flat, import with `from lib import x`

### Debugging Checklist
1. Check Kodi log: `~/.kodi/temp/kodi.log`
2. Verify addon.xml version bumped
3. Check Python syntax: `python -m py_compile addon.py`
4. Test imports: `python -c "from lib import x"`
5. Validate XML: `xmllint --noout addon.xml`
```

`scripts/qs.sh` (make executable: `chmod +x scripts/qs.sh`):

```bash
#!/bin/bash
# qs.sh - Quickstart: Generate AI context from system lenses
# Usage: qs.sh ["your query here"] [--compact|--deep]

set -euo pipefail

REPO_ROOT="${REPO:-$(pwd)}"
SYSTEM_DIR="$REPO_ROOT/system"
SHARE_DIR="$REPO_ROOT/share"

# Parse args
QUERY=""
DEPTH="standard"

while [[ $# -gt 0 ]]; do
    case $1 in
        --compact) DEPTH="compact"; shift ;;
        --deep) DEPTH="deep"; shift ;;
        --) shift; QUERY="$*"; break ;;
        *) QUERY="$QUERY $1"; shift ;;
    esac
done

QUERY=$(echo "$QUERY" | xargs)
TIMESTAMP=$(date +%Y%m%d_%H%M%S)
OUTPUT_FILE="$SHARE_DIR/generated/context_$TIMESTAMP.md"

# Ensure directory exists
mkdir -p "$SHARE_DIR/generated"

{
# Load core
cat "$SYSTEM_DIR/core/identity.yaml"
echo ""
cat "$SYSTEM_DIR/core/constraints.yaml"
echo ""
cat "$SYSTEM_DIR/core/env.yaml"
echo ""

# Classify intent and load lenses
# (Simple version - grep for triggers)
load_lens() {
    local lens="$1"
    local file="$SYSTEM_DIR/lenses/${lens}.yaml"
    if [[ -f "$file" ]]; then
        echo "<!-- Lens: $lens -->"
        cat "$file"
        echo ""
    fi
}

# Score query against triggers
SCORE_CODE=0
SCORE_DESIGN=0
SCORE_META=0

if echo "$QUERY" | grep -qiE "(fix|bug|error|def |class |import |refactor|debug|traceback|python|kodi)"; then
    SCORE_CODE=$((SCORE_CODE + 2))
fi
if echo "$QUERY" | grep -qiE "(idea|concept|design|structure|brainstorm|how might|what if|architecture)"; then
    SCORE_DESIGN=$((SCORE_DESIGN + 2))
fi
if echo "$QUERY" | grep -qiE "(continue|handoff|status|summary|last time|where were we|workflow|setup|init)"; then
    SCORE_META=$((SCORE_META + 2))
fi

# Load active lenses
[[ $SCORE_CODE -gt 1 ]] && load_lens "code"
[[ $SCORE_DESIGN -gt 1 ]] && load_lens "design"
[[ $SCORE_META -gt 1 ]] && load_lens "meta"

# Default to code if nothing triggered
[[ $SCORE_CODE -eq 0 && $SCORE_DESIGN -eq 0 && $SCORE_META -eq 0 ]] && load_lens "code"

# Load recent handoffs (based on depth)
HANDOFF_COUNT=1
[[ "$DEPTH" == "deep" ]] && HANDOFF_COUNT=3
[[ "$DEPTH" == "compact" ]] && HANDOFF_COUNT=0

if [[ $HANDOFF_COUNT -gt 0 ]]; then
    echo "<!-- Recent Context -->"
    ls -t "$SHARE_DIR/handoffs/"* 2>/dev/null | head -n $HANDOFF_COUNT | while read -r f; do
        echo ""
        echo "--- Handoff: $(basename "$f") ---"
        head -30 "$f"
    done
fi

# User query
echo ""
echo "<!-- User Query -->"
echo "Query: $QUERY"
echo "Depth: $DEPTH"

} > "$OUTPUT_FILE"

echo "Context generated: $OUTPUT_FILE"
cat "$OUTPUT_FILE"
```

`scripts/lib/intent_classifier.sh`:

```bash
#!/bin/bash
# intent_classifier.sh - Detect intent from query
# Usage: source intent_classifier.sh; classify_intent "query"

REPO_ROOT="${REPO:-$(pwd)}"
CLASSIFIER_FILE="$REPO_ROOT/system/router/classifier.yaml"

classify_intent() {
    local query="${1,,}"  # lowercase
    local active_lenses=()
    
    # Simple grep-based scoring (can be enhanced later)
    local score_code=0
    local score_design=0
    local score_meta=0
    
    # Code signals
    if [[ "$query" =~ (fix|bug|error|def |class |import |refactor|debug|traceback|python|kodi|addon) ]]; then
        score_code=$((score_code + 2))
    fi
    
    # Design signals
    if [[ "$query" =~ (idea|concept|design|structure|brainstorm|how might|what if|architecture|pattern) ]]; then
        score_design=$((score_design + 2))
    fi
    
    # Meta signals
    if [[ "$query" =~ (continue|handoff|status|summary|last time|where were we|workflow|process|setup|init) ]]; then
        score_meta=$((score_meta + 2))
    fi
    
    # Build result
    [[ $score_code -gt 1 ]] && active_lenses+=("code")
    [[ $score_design -gt 1 ]] && active_lenses+=("design")
    [[ $score_meta -gt 1 ]] && active_lenses+=("meta")
    
    # Default
    [[ ${#active_lenses[@]} -eq 0 ]] && active_lenses+=("code")
    
    echo "${active_lenses[*]}"
}

get_lens_path() {
    local lens="$1"
    echo "$REPO_ROOT/system/lenses/${lens}.yaml"
}
```
